	\section{Citing Some Books}

		\cite{Knuth-ACP} to learn more.
	\begin{btSect}[plain]{books}

		\btPrintCited
	\end{btSect}
	
	\section{Citing Some Articles}

	\cite{Knuth-LP} to learn more.
	\begin{btSect}[alpha]{articles}
%		\section*{Knuth's Articles}
		\btPrintCited
	\end{btSect}
