	\section{Blocks}
	\frame{\tableofcontents[currentsection]}
	{ % begin group
	%% Using Templates
	% 1. Setting the Color
	\setbeamercolor{frametitle}{fg=green, bg=teal}
	% 2. Seting the Font
	\setbeamerfont{frametitle}{series=\bfseries}
	% 3. Setting the overall Temp.
	\setbeamertemplate{frametitle}{
		\vspace{-1.5cm}
		\begin{center}
			\{\insertframetitle\}
		\end{center}			
	}
	% Setting the style of the block envs
	\setbeamertemplate{blocks}[default]
	\setbeamercolor{block title}{
		fg=black, bg=gray
	}
	\begin{frame}{Block Elements}
		\setbeamercovered{transparent}
		% The block env
		\begin{block}<1>{A Chunk of Material}
			Put some content here.........
		\end{block}
		% The alert block
		\begin{alertblock}<2>{The Alert Block}
			Put some content here.........
		\end{alertblock}
		% The example block
		\begin{exampleblock}<3>{Example Block}
			Some example content goes here....
		\end{exampleblock}
	\end{frame}
	} % end group
	
	\AtBeginEnvironment{theorem}{
		\setbeamercolor{block title}{bg=brown}
	}
	\begin{frame}{Theorem-Like Environments}
		\begin{theorem}[The Pythagorean Theorem]
			For a right triangle with legs of length 					$a$ and $b$ and hypotenuse of length $c$, 					we have: $a^2 + b^2 = c^2$.
		\end{theorem} 
		% Proof
		\AtBeginEnvironment{proof}{ 
   			\setbeamercolor{block title}{bg=olive}
		}
		\setbeamertemplate{qed symbol}{\checkmark}
		\setbeamercolor{qed symbol}{fg=olive}
		\begin{proof}
    		Proof body.........
  		\end{proof}
	\end{frame}
	
	\begin{frame}{Itemize and Enumerate}
		% Itemize 
		\setbeamercolor{block title}{bg=teal!80}
		\setbeamerfont{block title}{size=\Large}
		\setbeamertemplate{itemize item}[circle]
		\setbeamercolor{local structure}{fg=teal!80}
		\begin{block}{Some Itemize List}
			\begin{itemize}
				\item Item No. One 
				\item Item No. Two 
				\item Item No. Three
				\item[\ding{225}] PiFont Item.
			\end{itemize}
		\end{block}	
		% Enumerate
		\setbeamertemplate{enumerate item}[square]
		\setbeamercolor{local structure}{fg=black!80}
		\begin{enumerate}
			\item Item No. One
			\item Item No. Two
			\item Item No. Three
		\end{enumerate}			
	\end{frame}
