\documentclass{article}

\usepackage{kantlipsum, graphicx, xcolor}

\title{\huge\bf User Defined Environments}
\date{}

\newenvironment{mypar}{
	\vspace{0.5cm}
	% Upper Rule
	\noindent \textcolor{gray}{\rule{\textwidth}{3pt}}
	\vspace{1cm}
	
	{\Huge \textsc{T}}
	\Large \hspace{-0.7cm}
}{
	\vspace{1cm}
	\noindent \rule{\textwidth}{3pt} % \rule{width}{ht}
	\vspace{1cm}
}

\newenvironment{bfCenter}{
	\begin{center}
	\bfseries
	% \kant[1] -- The user input
}{
	\end{center}
}

\newenvironment{CNote}{
	\begin{center}
	\begin{minipage}{0.8\textwidth}
	\centering 
	\textbf{Note:} \hspace{-0.3cm}
}{
	\end{minipage}
	\end{center}
}

\newenvironment{items}{
	\begin{itemize}
	\bfseries
	\renewcommand{\labelitemi}{\textcircled{a}}
	\renewcommand{\labelitemii}{\dag}
	\renewcommand{\labelitemiii}{\textgreater}
	\providecommand{\bibtex}{bibtex}
}{
	\end{itemize}
}

\newenvironment{fig}[3]{
	\begin{figure}[#1]
	\centering
	\includegraphics[scale=#2]{#3}
}{
	\end{figure}
}

\newenvironment{hfig}[3][h]{
	\begin{figure}[#1]
	\centering
	\includegraphics[scale=#2]{#3}
}{
	\end{figure}
}
