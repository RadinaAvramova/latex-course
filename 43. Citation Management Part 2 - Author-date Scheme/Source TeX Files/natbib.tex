	\section{Basic Commands}
	\begin{description}
		\item[Textual Citation] \citet{Knuth}
		\item[Parenthetical Citations] \citep{Knuth}
		\item[Author Abbreviation] \citet{LGC97}
		\item[Full Author List] \citet*{LGC97}
		%% Notes [pre-notes][post-note]
		\item \rule{2cm}{2pt}
		\item[Post-Note] \citet[Chap.5]{Knuth}
		\item[Full-Note] \citep[see][Chap.5]{Knuth}
		\item[Pre-Note]  \citep[see][]{Knuth}
		%% Multiple Citations
		\item \rule{2cm}{2pt}
		\item[Different Authors] \citet{Knuth, TCM}
		\item[Same Author] \citet{TCM, TCM1}
	\end{description}
	
	\vspace{0.5cm}
	
	\section{Extended Commands}
	\begin{description}
		\item[Textual without Parentheses]
			\citealt{Knuth}
		\item[Parenthetical without Parentheses]
			\citealp{Knuth}
		\item[Citation Number] 
			\citenum{Knuth,TCM1,LGC97,TCM}
		\item \rule{2cm}{2pt}
		\item[Citing with Text] 
			\citetext{
			Refer to \citet{Knuth}, or check \citep{TCM} 
			}
		\item \rule{2cm}{2pt}
		\item[Citing Author Only] \citeauthor{Knuth}
		\item[Citing Year Only] 
			\citeyear{Knuth}, or \citeyearpar{TCM}
		\item \rule{2cm}{2pt}
		\defcitealias{Knuth}{
			\textcolor{orange}{The \TeX\,Book}}
		\item[Textual Aliasing] \citetalias{Knuth}
	\end{description}
