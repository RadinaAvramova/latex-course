  		F(x) = \begin{cases}

  					0 ~, & \text{Otherwise.}

	\end{equation}

	\subsection{Equation-Lining:}
	Underlining an Expression:
	\begin{equation}
		A = P + \left(
		\frac{2}{n+1} + \underline{\left[2\times m			\right]}
			\right)
	\end{equation}
	Over-lining an Expression:
	\begin{equation}
		A = P + \left(
		\frac{2}{n+1} + \overline{\left[2\times m			\right]}
			\right)
	\end{equation}
	
	Over and Under-braced Expressions:
	\begin{equation}
		A = \overbrace{B+C+D}^{ANote} \times
			\underbrace{E+F+G}_{DNote}
	\end{equation}
	
	\subsection{Split Equations:}
	\begin{equation}
		\begin{split}
		\lim_{x \rightarrow 0} \frac{\sin x}{x} =
		\lim_{x \rightarrow 0} \frac{e^x-1}{x} &=\\
		&\lim_{x \rightarrow 0} \frac{\ln(1+x)}{x} =
		\lim_{n \rightarrow \infty} \sqrt[n]{n} = 1
		\end{split}
	\end{equation}

	\subsection{Stacking and Chemical Equations:}
	\centering{
	$\overset{i}{x}$, \quad $\underset{j}{x}$ \\
	\quad And:
	$\overset{i}{\underset{j}{X}}$,
	is the same as:
	$\underset{j}{\overset{i}{X}}$
	\vspace{5mm} \ \\
	
	$\underset{i=0}{\overset{n}{\sum}}$ \\
	is the same as:
	\[\sum^n_{i=0}\]
	
	\begin{equation*} 
		\sum_{
			\substack{\text{$i$ odd} \\ i \to n}
		}^{
			\substack{\text{$j$ even} \\ j \to m}		
		}
	\end{equation*}	
	
	\begin{equation*}
		A \stackrel{to}{\to} B
	\end{equation*}		
	
	\begin{equation*}
		\mathrm{2Na + D_2}
		\stackrel{\mathrm{heat}}{\rightleftharpoons}
		\mathrm{2NaD}
	\end{equation*}

	
	\begin{equation*}
		\mathrm{H_2 + D_2}
		{}_{~\overrightarrow{\mathrm{Put \, Thing  \, 					here}~}}
		\mathrm{2HD}
	\end{equation*}

	\begin{equation*}
		\mathrm{H_2 + D_2}
		{}^{~\underrightarrow{\mathrm{Put \, Thing  \, 					here}~}}
